  configurationId: "39a4d283-ca87-4a8a-bdbb-df212eca5fdb",

  query: null,
  reviewerAssignmentPolicy: "MANUALLY",
  allowSelfReview: false,
  sendReviewerAssignmentNotification: true,
  allowReviewerReassignment: false,
  startDate: null,
  endDate: new Date("2026-09-30T00:00:00Z"),

  reminderSchedule: [
    7,
    3,
    1,
  ],
  reminderIncludeManager: true,
  requireReasonOnDenial: false,
  hideAiSuggestions: false,
  customStartMessage: null,
  groupAssetVisibilityPolicy: "STRICT",
  isTemplate: false,
  cronExpression: null,
  nextScheduledRun: null,
  lastScheduledRun: null,
  recurringDurationDays: null,
};

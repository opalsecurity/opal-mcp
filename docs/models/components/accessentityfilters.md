# AccessEntityFilters

Filters for matching entities by type, name, tag, IDs, connections, or access levels. Supports recursive logical composition via allOf/anyOf.

## Example Usage

```typescript
import { AccessEntityFilters } from "opal-mcp/models/components";

let value: AccessEntityFilters = {
  entityItemTypes: [
    "OPAL_ROLE",
  ],
  entityName: {
    stringMatchType: "REGEX",
    string: "engineering",
  },
  entityTag: {
    key: "team",
    value: "platform",
  },
  hrIdpStatus: {
    statuses: [
      "ACTIVE",
    ],
  },
  allOf: [
    {
      entityItemTypes: [
        "OPAL_ROLE",
      ],
      entityName: {
        stringMatchType: "REGEX",
        string: "engineering",
      },
      entityTag: {
        key: "team",
        value: "platform",
      },
      hrIdpStatus: {
        statuses: [
          "ACTIVE",
        ],
      },
      anyOf: [
        {
          entityItemTypes: [
            "OPAL_ROLE",
          ],
          entityName: {
            stringMatchType: "REGEX",
            string: "engineering",
          },
          entityTag: {
            key: "team",
            value: "platform",
          },
          hrIdpStatus: {
            statuses: [
              "ACTIVE",
            ],
          },
        },
      ],
    },
  ],
  anyOf: [
    {
      entityItemTypes: [
        "OPAL_ROLE",
      ],
      entityName: {
        stringMatchType: "REGEX",
        string: "engineering",
      },
      entityTag: {
        key: "team",
        value: "platform",
      },
      hrIdpStatus: {
        statuses: [
          "ACTIVE",
        ],
      },
    },
  ],
};
```
